const http = require('http'); const fs = require('fs'); const path = require('path'); const server = http.createServer((req, res) => { fs.readFile(path.join(__dirname, 'index.html'), (err, data) => { if (err) { res.writeHead(500, { 'Content-Type': 'text/plain; charset=utf-8' }); res.end('Server Error'); return; } res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' }); res.end(data); }); }); const PORT = process.env.PORT || 3000; server.listen(PORT, () => { console.log(`Server is running on port ${PORT}`); });
